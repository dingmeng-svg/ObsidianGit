# 📅 本月洞察

```dataviewjs
const now = dv.date("now");
const monthStart = now.startOf("month");
const files = dv.pages('"90_System/Insight"')
  .where(p => p.file.name.match(/^\d{4}-\d{2}-\d{2}-Insight$/))
  .where(p => p.file.day && p.file.day >= monthStart)
  .sort(p => p.file.day, "desc");
if (files.length > 0) {
  dv.list(files.map(p => p.file.link));
} else {
  dv.span("本月暂无洞察记录");
}
```
