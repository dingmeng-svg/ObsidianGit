> try {
>     const now = dv.luxon.DateTime.now();
>     const todayStr = now.toFormat("yyyy-MM-dd");
>     const monthStart = now.startOf("month");
>
>     const collectorFile = dv.page("00_Inbox/Collector");
>     const cTotal = collectorFile?.file?.tasks?.length || 0;
>     const cDone = collectorFile?.file?.tasks?.where(t => t.completed).length || 0;
>
>     const todayDiary = dv.pages('"20_Areas/A2_能力接口/Bullet Journal/Daily"')
>         .where(p => p.file.day && p.file.day.toFormat("yyyy-MM-dd") === todayStr)
>         .first();
>
>     let tDone = 0, tTotal = 0, hDone = 0, hTotal = 0;
>     if (todayDiary?.file?.tasks) {
>         const tasks = todayDiary.file.tasks;
>         const validTasks = tasks.where(t => {
>             const text = t.text.trim();
>             // 排除空行、纯checkbox、习惯打卡、训练区块
>             if (text.length === 0 || text === "- [ ]" || text === "- [x]") return false;
>             if (text.includes("[habit::")) return false;
>             if (text.includes("上交叉矫正训练") || t.header?.subpath?.includes("上交叉矫正")) return false;
>             return true;
>         });
>         tTotal = validTasks.length;
>         tDone = validTasks.where(t => t.completed).length;
>         const hTasks = tasks.where(t => t.text.includes("[habit::"));
>         hTotal = hTasks.length;
>         hDone = hTasks.where(t => t.completed).length;
>     }
>
>     const mDays = Math.floor(now.diff(monthStart, "days").days) + 1;
>     const mDiaries = dv.pages('"20_Areas/A2_能力接口/Bullet Journal/Daily"')
>         .where(p => p.file.name.match(/^\d{4}-\d{2}-\d{2}-Daily-Log$/) && p.file.day && p.file.day >= monthStart).length;
>
>     const rawTotal = dv.pages('"00_Inbox/_raw"').where(p => p.file.name != "_raw MOC").length;
>     const rawPending = dv.pages('"00_Inbox/_raw"').where(p => p.file.name != "_raw MOC" && p.ingest_status != "ingested").length;
>     const extTotal = dv.pages('"00_Inbox/外部导入"').where(p => p.file.name != "_说明").length;
>     const extPending = dv.pages('"00_Inbox/外部导入"').where(p => p.file.name != "_说明" && p.status != "archived").length;
>     const iTotal = rawTotal + extTotal;
>     const iPending = rawPending + extPending;
>
>     const stats = [
>         { label: "📬 Nexus", val: `${cDone}/${cTotal}` },
>         { label: "🎯 Tasks", val: `${tDone}/${tTotal}` },
>         { label: "☑️ Habits", val: `${hDone}/${hTotal}` },
>         { label: "📝 Daily", val: `${mDiaries}/${mDays}` },
>         { label: "📥 Inbox", val: `${iPending}/${iTotal}` }
>     ];
>
>     let pills = `<div class="ab-stats-wrap">`;
>     stats.forEach(s => {
>         const parts = String(s.val).split('/');
>         const current = parts[0] || '0';
>         const total = parts[1] || '0';
>         pills += `
>             <div class="ab-pill">
>                 <span class="ab-pill-label">${s.label}</span>
>                 <span class="ab-pill-val"><span class="ab-current">${current}</span><span class="ab-total">/${total}</span></span>
>             </div>
>         `;
>     });
>     pills += `</div>`;
>     dv.span(pills);
>
> } catch (e) {
>     dv.span(`<span class="ab-err">⚠️ 数据加载异常</span>`);
