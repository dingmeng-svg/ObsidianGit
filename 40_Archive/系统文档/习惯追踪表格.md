```dataview

TABLE WITHOUT ID
  file.day AS "日期",
  choice(workout, "✅", "⚪") AS "运动",
  choice(read, "✅", "⚪") AS "阅读",
  choice(meditate, "✅", "⚪") AS "冥想"
FROM "日记文件夹"
WHERE file.day >= date(2026-06-01) AND file.day <= date(2026-06-30)
SORT file.day ASC
```
