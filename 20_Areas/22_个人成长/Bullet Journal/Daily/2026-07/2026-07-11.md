    .where(p => p.file.day && p.file.day.ts === yesterday.ts)
    .first();

    let tasks = yesterdayPage.file.tasks
        .where(t => !t.completed && t.header.subpath.includes("今日闪念捕获"));
    if (tasks.length) {
        dv.taskList(tasks);

    }
} else {
    dv.paragraph("📭 未找到昨天的日记文件。");
}
