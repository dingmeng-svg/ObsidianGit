# 配色对比测试：三种绿色系

## 配色 A：当前活跃（apple-alpha-green）

```dataviewjs
const colorsA = ["#F4FCF6", "#CCF1D5", "#99E3AC", "#67D583", "#34C759"];

function renderBar(label, pct, colorArr, name) {
    let idx = 0;
    if (pct > 60) idx = 4;
    else if (pct > 45) idx = 3;
    else if (pct > 30) idx = 2;
    else if (pct > 15) idx = 1;
    const bg = colorArr[idx];
    const textColor = idx < 2 ? "#34C759" : "#fff";
    return `<div style="display:flex;align-items:center;gap:12px;margin-bottom:8px;">
        <span style="min-width:100px;font-size:13px;color:#4B5563;text-align:right;flex-shrink:0;">${label}</span>
        <div style="flex:1;max-width:460px;height:8px;background:#F3F4F6;border-radius:99px;overflow:hidden;">
            <div style="width:${pct}%;height:100%;border-radius:99px;background:${bg};"></div>
        </div>
        <span style="min-width:30px;font-size:13px;font-weight:600;color:#4B5563;">${bg}</span>
    </div>`;
}

let html = `<div style="margin:1rem 0;font-family:-apple-system,sans-serif;">`;
html += `<h3 style="font-size:16px;font-weight:600;margin-bottom:12px;">配色 A：apple-alpha-green（当前）</h3>`;
[100, 60, 35, 20, 5].forEach(p => {
    html += renderBar(`样例 ${p}%`, p, colorsA);
});
html += `<div style="display:flex;gap:4px;margin-top:8px;">`;
colorsA.forEach(c => {
    html += `<div style="flex:1;height:20px;border-radius:4px;background:${c};display:flex;align-items:center;justify-content:center;"><span style="font-size:9px;color:${c === "#F4FCF6" || c === "#CCF1D5" ? "#333" : "#fff"};">${c}</span></div>`;
});
html += `</div></div>`;
dv.span(html);
```
